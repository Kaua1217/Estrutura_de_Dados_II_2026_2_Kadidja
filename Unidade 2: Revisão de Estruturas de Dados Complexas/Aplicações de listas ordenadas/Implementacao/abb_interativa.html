class No:
    # Cada nó guarda um valor e duas referências:
    # uma para a subárvore esquerda e outra para a direita.
    def __init__(self, valor):
        self.valor = valor
        self.esquerda = None
        self.direita = None


def inserir(raiz, valor):
    # Se chegou numa posição vazia, cria o nó aqui.
    if raiz is None:
        return No(valor)

    # Valor menor que o nó atual: desce para a esquerda.
    if valor < raiz.valor:
        raiz.esquerda = inserir(raiz.esquerda, valor)
    # Valor maior (ou igual): desce para a direita.
    else:
        raiz.direita = inserir(raiz.direita, valor)

    return raiz


def buscar(raiz, alvo):
    # "atual" avança pela árvore, um nó por vez.
    atual = raiz

    while atual is not None:
        # Achou: o valor procurado é o nó atual.
        if alvo == atual.valor:
            return atual

        # Alvo menor: só pode estar à esquerda.
        if alvo < atual.valor:
            atual = atual.esquerda
        # Alvo maior: só pode estar à direita.
        else:
            atual = atual.direita

    # Saiu do laço sem achar: caiu em None.
    return None


# --- Montando a árvore do exercício ---
valores = [50, 30, 70, 20, 40, 60, 80]
raiz = None
for v in valores:
    raiz = inserir(raiz, v)

# --- Buscando um valor ---
resultado = buscar(raiz, 60)
print(resultado.valor if resultado else "não encontrado")
